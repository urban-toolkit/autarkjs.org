<script setup>
const mainCode = `
import { ComputeRender } from '@urban-toolkit/autk-compute'
import { ColorMapDomainStrategy, TriangulatorBuildingWithWindows } from '@urban-toolkit/autk-core'
import { AutkDb } from '@urban-toolkit/autk-db'
import { AutkMap } from '@urban-toolkit/autk-map'
import { AutkPlot } from '@urban-toolkit/autk-plot'

const BUILDINGS_LAYER = 'table_osm_buildings'
const GENERATED_LAYER_ID = 'selected_building_windows'
const ANALYSIS_FLOORS = 10

setStatus('Initializing spatial database...')
const db = new AutkDb()
await db.init()

setStatus('Loading OSM layers from the local PBF...')
await db.loadOsm({
  pbfFileUrl: '/data/lower_mnt.osm.pbf',
  queryArea: {
    geocodeArea: 'New York',
    areas: ['Battery Park City', 'Financial District'],
  },
  outputTableName: 'table_osm',
  autoLoadLayers: {
    layers: ['surface', 'parks', 'water', 'roads', 'buildings'],
  },
})

const buildings = await db.getLayer(BUILDINGS_LAYER)
const map = new AutkMap(canvas)

setStatus('Initializing map and dashboard...')
await map.init()
for (const layerData of db.getLayersMetadata()) {
  const geojson = layerData.name === BUILDINGS_LAYER ? buildings : await db.getLayer(layerData.name)
  map.loadCollection(layerData.name, { collection: geojson, type: layerData.type })
}

const chart = new AutkPlot(plot, {
  type: 'parallel-coordinates',
  collection: buildWindowMetricsCollection([]),
  attributes: { axis: ['parksVisibility', 'waterVisibility', 'skyVisibility', 'viewScore'] },
  labels: {
    axis: ['Parks', 'Water', 'Sky', 'Score'],
    title: 'Selected building window metrics',
  },
  margins: { left: 20, right: 20, top: 100, bottom: 15 },
  width: plot.clientWidth || 900,
  height: 420,
})

map.updateRenderInfo(BUILDINGS_LAYER, { isPick: true, opacity: 0.94 })
map.draw()
clearStatus()

attachPickingHandler({
  map,
  db,
  chart,
  buildings,
  BUILDINGS_LAYER,
  GENERATED_LAYER_ID,
  ANALYSIS_FLOORS,
})
`

const listenersCode = `
function attachPickingHandler(params) {
  const { map, db, chart, buildings, BUILDINGS_LAYER, GENERATED_LAYER_ID, ANALYSIS_FLOORS } = params
  let analysisVersion = 0

  map.events.on(MapEvent.PICKING, async ({ selection, layerId }) => {
    if (layerId !== BUILDINGS_LAYER) return

    if (selection.length === 0) {
      analysisVersion += 1
      clearSelectionState(map, chart)
      return
    }

    const pickedId = selection[selection.length - 1]
    if (selection.length > 1) {
      map.setHighlightedIds(BUILDINGS_LAYER, [pickedId])
    }

    const requestId = ++analysisVersion
    const pickedBuilding = buildings.features[pickedId]
    if (!pickedBuilding) {
      clearSelectionState(map, chart)
      return
    }

    map.clearSkippedIds(BUILDINGS_LAYER)
    map.setSkippedIds(BUILDINGS_LAYER, [pickedId])
    map.removeLayer(GENERATED_LAYER_ID)

    const pickedCollection = {
      type: 'FeatureCollection',
      features: [pickedBuilding],
    }

    const layout = TriangulatorBuildingWithWindows.buildWindowLayout(pickedCollection, ANALYSIS_FLOORS)
    if (layout.collection.features.length === 0) {
      clearSelectionState(map, chart)
      return
    }

    setStatus('Computing selected building view metrics...')
    const parksGeoJson = await db.getLayer('table_osm_parks')
    const waterGeoJson = await db.getLayer('table_osm_water')
    const sceneBuildings = {
      ...buildings,
      features: buildings.features.filter((_, index) => index !== pickedId),
    }

    const render = new ComputeRender()
    const windowScores = await render.run({
      layers: [
        { id: BUILDINGS_LAYER, collection: sceneBuildings, type: 'buildings' },
        { id: 'table_osm_parks', collection: parksGeoJson, type: 'parks' },
        { id: 'table_osm_water', collection: waterGeoJson, type: 'water' },
      ],
      viewpoints: {
        collection: pickedCollection,
        strategy: { type: 'building-windows', floors: ANALYSIS_FLOORS },
      },
      aggregation: { type: 'classes', includeBackground: true, backgroundLayerType: 'sky' },
      tileSize: 32,
    })

    if (requestId !== analysisVersion) return

    const thematicByWindowId = new Map()
    const plotFeatures = []

    for (const feature of windowScores.features) {
      const windowId = String(feature.properties?.windowId ?? '')
      if (!windowId) continue

      const renderMetrics = feature.properties?.compute?.render ?? {}
      const classes = renderMetrics.classes ?? {}
      const parksVisibility = Number(classes.parks ?? 0)
      const waterVisibility = Number(classes.water ?? 0)
      const skyVisibility = Number(classes.sky ?? 0)
      const viewScore = waterVisibility * 0.2 + parksVisibility * 0.35 + skyVisibility * 0.45

      thematicByWindowId.set(windowId, {
        parksVisibility,
        waterVisibility,
        skyVisibility,
        viewScore,
      })

      plotFeatures.push({
        type: 'Feature',
        geometry: null,
        properties: {
          windowIndex: plotFeatures.length,
          parksVisibility,
          waterVisibility,
          skyVisibility,
          viewScore,
        },
      })
    }

    const mesh = buildAnalysisMesh({
      collection: pickedCollection,
      origin: map.layerManager.origin,
      floors: ANALYSIS_FLOORS,
      thematicByWindowId,
    })

    if (!mesh) {
      clearSelectionState(map, chart)
      return
    }

    map.loadMesh(GENERATED_LAYER_ID, mesh)
    map.updateColorMap(GENERATED_LAYER_ID, {
      colorMap: {
        domainSpec: { type: ColorMapDomainStrategy.PERCENTILE, params: [5, 95] },
      },
    })
    map.updateRenderInfo(GENERATED_LAYER_ID, { isColorMap: true, opacity: 1.0 })

    chart.updateCollection(buildWindowMetricsCollection(plotFeatures))
    map.draw()
    clearStatus()
  })
}
`

const helpersCode = `
function clearSelectionState(map, chart) {
  map.clearSkippedIds('table_osm_buildings')
  map.removeLayer('selected_building_windows')
  chart.updateCollection(buildWindowMetricsCollection([]))
  map.draw()
}

function buildWindowMetricsCollection(features) {
  return {
    type: 'FeatureCollection',
    features: features.length > 0
      ? features
      : [
          {
            type: 'Feature',
            geometry: null,
            properties: {
              windowIndex: 0,
              parksVisibility: 0,
              waterVisibility: 0,
              skyVisibility: 0,
              viewScore: 0,
            },
          },
        ],
  }
}

function buildAnalysisMesh(params) {
  const [geometry, components] = TriangulatorBuildingWithWindows.buildMesh(
    params.collection,
    params.origin,
    params.floors,
  )
  if (geometry.length === 0 || components.length === 0) return null

  const thematic = components.map((component) => {
    if (!component.featureId) return { value: 0, valid: 0 }
    const score = params.thematicByWindowId.get(String(component.featureId))?.viewScore ?? 0
    return { value: score, valid: 1 }
  })

  return { geometry, components, thematic, type: 'buildings' }
}
`

# View Score

Pick a building, compute per-window visibility metrics, render the result in 3D, and compare the windows in a linked parallel coordinates plot. This example combines map, compute, and plot workflows in a single analysis view.
