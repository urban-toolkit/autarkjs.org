<script setup>
const code = `
import { AutkDb } from '@urban-toolkit/autk-db'
import { AutkMap, MapEvent } from '@urban-toolkit/autk-map'
import { AutkPlot, PlotEvent } from '@urban-toolkit/autk-plot'

output('<div style="height: 340px;"><div id="plotBody" style="width: 100%; height: 100%;"></div></div>')
const plotDiv = mount.querySelector('#plotBody')

setStatus('Initializing spatial database...')
const db = new AutkDb()
await db.init()

setStatus('Loading road and event datasets...')
await db.loadGeojson({
  geojsonFileUrl: '/data/mnt_roads.geojson',
  outputTableName: 'roads',
  coordinateFormat: 'EPSG:3395'
})
await db.loadCsv({
  csvFileUrl: '/data/noise_manhattan_clean.csv',
  outputTableName: 'noise',
  geometryColumns: {
    latColumnName: 'latitude',
    longColumnName: 'longitude',
    coordinateFormat: 'EPSG:3395',
  },
})

setStatus('Joining events to nearby roads...')
await db.spatialQuery({
  tableRootName: 'roads',
  tableJoinName: 'noise',
  near: { distance: 200 },
  groupBy: [
    { column: 'key', aggregateFn: 'count' },
    { column: 'date', aggregateFn: 'collect' },
  ],
})

const roads = await db.getLayer('roads')

setStatus('Initializing linked views...')
const map = new AutkMap(canvas)
await map.init()
for (const layerData of db.getLayersMetadata()) {
  const geojson = await db.getLayer(layerData.name)
  map.loadCollection(layerData.name, { collection: geojson, type: layerData.type })
}
map.updateThematic('roads', { collection: roads, property: 'properties.sjoin.count.noise' })
map.updateRenderInfo('roads', { isPick: true, isColorMap: true })
map.draw()

let selectionSource = null

const plot = new AutkPlot(plotDiv, {
  type: 'linechart',
  collection: roads,
  attributes: { axis: ['sjoin.collect.noise', '@transform'] },
  labels: { axis: ['Time bucket', 'Event count'], title: 'Monthly noise events per road' },
  transform: {
    preset: 'binning-events',
    options: {
      timestamp: 'date',
      resolution: 'day',
      reducer: 'count',
    },
  },
  margins: { left: 60, right: 20, top: 50, bottom: 140 },
  width: plotDiv.clientWidth || 900,
  events: [PlotEvent.BRUSH_X],
})

plot.events.on(PlotEvent.BRUSH_X, ({ selection }) => {
  if (selectionSource === 'map') return
  selectionSource = 'plot'
  map.setHighlightedIds('roads', selection)
  selectionSource = null
})

map.events.on(MapEvent.PICKING, ({ selection }) => {
  if (selectionSource === 'plot') return
  selectionSource = 'map'
  plot.setSelection(selection)
  selectionSource = null
})

clearStatus()
`
</script>

# Road Event Timeline

Join complaint events to nearby roads, color the street network by event count, and explore the temporal distribution with a linked line chart. This example highlights spatial joins, time-based aggregation, and brush-driven linked analysis.
