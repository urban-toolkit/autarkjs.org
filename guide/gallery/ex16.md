const mainCode = `
import { AutkDb } from '@urban-toolkit/autk-db'
import { ComputeRender } from '@urban-toolkit/autk-compute'
import { ColorMapDomainStrategy } from '@urban-toolkit/autk-core'
import { AutkMap, MapEvent } from '@urban-toolkit/autk-map'

const BUILDINGS_LAYER = 'table_osm_buildings'

setStatus('Initializing spatial database...')
const db = new AutkDb()
await db.init()

setStatus('Loading OSM layers from the local PBF...')
await db.loadOsm({
  pbfFileUrl: '/data/lower_mnt.osm.pbf',
  queryArea: {
    geocodeArea: 'New York',
    areas: ['Battery Park City', 'Financial District'],
  },
  outputTableName: 'table_osm',
  autoLoadLayers: {
    layers: ['surface', 'parks', 'water', 'roads', 'buildings'],
  },
})

const buildings = await db.getLayer(BUILDINGS_LAYER)
const buildingsForCompute = {
  ...buildings,
  features: buildings.features.map((feature, index) => ({
    ...feature,
    properties: {
      ...feature.properties,
      _renderObjectId: index,
    },
  })),
}

let buildingsWithVisibility = resetVisibility(buildings)

setStatus('Initializing map...')
const map = new AutkMap(canvas)
await map.init()
for (const layerData of db.getLayersMetadata()) {
  const geojson = layerData.name === BUILDINGS_LAYER ? buildingsWithVisibility : await db.getLayer(layerData.name)
  map.loadCollection(layerData.name, { collection: geojson, type: layerData.type })
}
map.updateColorMap(BUILDINGS_LAYER, {
  colorMap: {
    domainSpec: { type: ColorMapDomainStrategy.MIN_MAX },
  },
})
map.updateThematic(BUILDINGS_LAYER, {
  collection: buildingsWithVisibility,
  property: 'properties.compute.visibilityValue',
})
map.updateRenderInfo(BUILDINGS_LAYER, { isColorMap: true, isPick: true, opacity: 0.92 })
map.draw()
clearStatus()

map.events.on(MapEvent.PICKING, async ({ selection, layerId }) => {
  if (layerId !== BUILDINGS_LAYER) return

  if (selection.length === 0) {
    buildingsWithVisibility = resetVisibility(buildings)
    refreshBuildingsLayer(map, buildingsWithVisibility)
    return
  }

  const pickedId = selection[selection.length - 1]
  const pickedFeature = buildingsForCompute.features[pickedId]
  if (!pickedFeature) return

  const footprint = resolveBuildingFootprint(pickedFeature)
  if (!footprint) return

  setStatus('Computing building visibility...')
  const height = resolveBuildingHeight(pickedFeature)
  const viewpoints = buildVerticalViewpoints(footprint, height)
  const sceneBuildings = {
    ...buildingsForCompute,
    features: buildingsForCompute.features.filter((_, index) => index !== pickedId),
  }

  const render = new ComputeRender()
  const pointResults = await render.run({
    layers: [{
      id: BUILDINGS_LAYER,
      collection: sceneBuildings,
      type: 'buildings',
      objectIdProperty: '_renderObjectId',
    }],
    viewpoints: {
      collection: viewpoints,
      sampling: { directions: 36 },
    },
    aggregation: { type: 'objects' },
    tileSize: 64,
  })

  const visibilityScores = new Map()
  for (const feature of pointResults.features) {
    const objects = feature.properties?.compute?.render?.objects ?? {}
    for (const key of Object.keys(objects)) {
      const metric = objects[key]
      if (!metric?.visible) continue
      const id = parseObjectKey(key)
      if (id !== null && id !== pickedId) {
        const score = Number(metric.sampleRatio ?? 0)
        visibilityScores.set(id, (visibilityScores.get(id) ?? 0) + score)
      }
    }
  }

  const pointCount = Math.max(1, pointResults.features.length)
  buildingsWithVisibility = {
    ...buildings,
    features: buildings.features.map((feature, index) => ({
      ...feature,
      properties: {
        ...feature.properties,
        compute: {
          ...(feature.properties?.compute ?? {}),
          visibility: (visibilityScores.get(index) ?? 0) > 0,
          visibilityScore: (visibilityScores.get(index) ?? 0) / pointCount,
          visibilityValue: (visibilityScores.get(index) ?? 0) / pointCount,
        },
      },
    })),
  }

  refreshBuildingsLayer(map, buildingsWithVisibility)
  clearStatus()
})
`

const helpersCode = `
function resetVisibility(buildings) {
  return {
    ...buildings,
    features: buildings.features.map((feature) => ({
      ...feature,
      properties: {
        ...feature.properties,
        compute: {
          ...(feature.properties?.compute ?? {}),
          visibility: false,
          visibilityScore: 0,
          visibilityValue: 0,
        },
      },
    })),
  }
}

function refreshBuildingsLayer(map, buildingsWithVisibility) {
  map.updateThematic('table_osm_buildings', {
    collection: buildingsWithVisibility,
    property: 'properties.compute.visibilityValue',
  })
  map.updateRenderInfo('table_osm_buildings', { isColorMap: true, isPick: true })
  map.draw()
}

function buildVerticalViewpoints(footprint, height) {
  const [x, y] = barycenter(footprint)
  const floorHeight = 3.4
  const sampleCount = Math.max(1, Math.round(height / floorHeight))
  const features = []

  for (let i = 0; i < sampleCount; i++) {
    const z = sampleCount === 1 ? Math.min(height, 1.7) : (height * i) / (sampleCount - 1)
    features.push({
      type: 'Feature',
      geometry: { type: 'Point', coordinates: [x, y, z] },
      properties: { level: i, z },
    })
  }

  return { type: 'FeatureCollection', features }
}

function barycenter(ring) {
  let x = 0
  let y = 0
  for (const [px, py] of ring) {
    x += px
    y += py
  }
  return [x / ring.length, y / ring.length]
}

function resolveBuildingFootprint(feature) {
  const geometry = feature.geometry
  if (!geometry) return null

  let bestRing = null
  let bestArea = -1
  const consider = (ring) => {
    if (!ring || ring.length < 3) return
    const area = Math.abs(computeRingArea(ring))
    if (area > bestArea) {
      bestArea = area
      bestRing = ring
    }
  }

  const scan = (geom) => {
    if (geom.type === 'Polygon') {
      consider(geom.coordinates[0])
      return
    }
    if (geom.type === 'MultiPolygon') {
      for (const polygon of geom.coordinates) consider(polygon[0])
    }
  }

  if (geometry.type === 'GeometryCollection') {
    for (const part of geometry.geometries) scan(part)
  } else {
    scan(geometry)
  }

  return bestRing
}

function computeRingArea(ring) {
  let area = 0
  for (let i = 0; i < ring.length; i++) {
    const [x1, y1] = ring[i]
    const [x2, y2] = ring[(i + 1) % ring.length]
    area += x1 * y2 - x2 * y1
  }
  return area * 0.5
}

function resolveBuildingHeight(feature) {
  const rootProps = feature.properties ?? {}
  const parts = Array.isArray(rootProps.parts) ? rootProps.parts : []

  const parseHeight = (props) => {
    if (!props) return null
    const rawHeight = parseFloat(String(props.height ?? props['building:height'] ?? ''))
    const rawLevels = parseFloat(String(props['building:levels'] ?? props.levels ?? '')) * 3.4
    const value = Number.isFinite(rawHeight) && rawHeight > 0 ? rawHeight : Number.isFinite(rawLevels) && rawLevels > 0 ? rawLevels : NaN
    return Number.isFinite(value) && value > 0 ? value : null
  }

  const partHeights = parts.map((part) => parseHeight(part)).filter((value) => value !== null)
  if (partHeights.length > 0) return Math.max(...partHeights)
  return parseHeight(rootProps) ?? 20
}

function parseObjectKey(key) {
  const separator = key.indexOf(':')
  if (separator < 0) return null
  const raw = decodeURIComponent(key.slice(separator + 1))
  const parsed = Number.parseInt(raw, 10)
  return Number.isFinite(parsed) ? parsed : null
}
`

# Visibility

Pick a building and compute which surrounding buildings become visible from vertically sampled viewpoints. This example highlights object-based visibility analysis, interactive building selection, and compute-driven thematic updates.
